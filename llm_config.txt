# =============================================================================
# Backend selection
# =============================================================================
# BACKEND=transformers
#   Uses HuggingFace Transformers + bitsandbytes 4-bit quantization.
#   Recommended for single-GPU runs on small/mid models (≤ 31 B).
#   Requires: transformers, bitsandbytes, lmformatenforcer, accelerate.
#
# BACKEND=vllm
#   Uses vLLM for high-throughput, multi-GPU inference.
#   Enables: Automatic Prefix Caching (APC), tensor parallelism,
#   native xgrammar-based guided JSON decoding (no lmformatenforcer needed),
#   and batched document processing.
#   Recommended for all models ≥ 70 B and any multi-GPU node.
#   Requires: vllm (pip install vllm)
#
BACKEND=transformers

# =============================================================================
# vLLM-specific settings (ignored when BACKEND=transformers)
# =============================================================================
#
# TENSOR_PARALLEL_SIZE
#   Number of GPUs to shard the model across.
#   Set to 1 for single-GPU, 2 for two 80-GB GPUs, 4 for four GPUs.
#   Rule of thumb: use the model's "recommended_tp" value from the registry.
#
TENSOR_PARALLEL_SIZE=1

# GPU_MEMORY_UTILIZATION
#   Fraction of each GPU's VRAM that vLLM may use for the KV cache.
#   0.90 is the recommended default. Lower to 0.80 if you get OOM on very
#   long documents. Do not exceed 0.95.
#
GPU_MEMORY_UTILIZATION=0.90

# GUIDED_DECODING_BACKEND
#   xgrammar  — fastest, grammar-based, recommended (default)
#   outlines  — alternative, slightly slower
#
GUIDED_DECODING_BACKEND=xgrammar

# ENABLE_PREFIX_CACHING
#   true  — Automatic Prefix Caching (APC). The system prompt (which embeds
#            the full TEATER vocabulary) is computed once and its KV-cache is
#            reused for every line in the document. This is the main
#            throughput multiplier for this pipeline. Recommended for all runs.
#   false — Disable APC. Only useful for debugging or very short documents
#            where cache overhead outweighs the benefit.
#
ENABLE_PREFIX_CACHING=true

# VLLM_BATCH_SIZE
#   Number of document lines submitted to vLLM in a single generate() call.
#   Larger batches = higher GPU utilisation but more memory pressure.
#   16 is a safe default; increase to 32–64 on nodes with ≥ 160 GB VRAM.
#   Consecutive-error abort logic is evaluated after each mini-batch.
#
VLLM_BATCH_SIZE=16

# MAX_MODEL_LEN (optional override)
#   Override the model's default maximum sequence length.
#   Leave commented to use the model's native context window.
#   Set lower (e.g. 65536) to reduce KV-cache memory on long-context models.
#
# MAX_MODEL_LEN=65536
